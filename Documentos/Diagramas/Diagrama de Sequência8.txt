@startuml

actor Cliente

participant "Front/Rota (GET /produtos)" as Rota
participant ProdutoController
participant ProdutoService
participant ProdutoRepository
database BD

Cliente -> Rota : listarProdutos()

Rota -> ProdutoController : getProdutos(req, res)

ProdutoController -> ProdutoService : listarTodos()

ProdutoService -> ProdutoRepository : findAll()

ProdutoRepository -> BD : SELECT * FROM produto
BD --> ProdutoRepository : listaProdutos

ProdutoRepository --> ProdutoService : listaProdutos

ProdutoService --> ProdutoController : listaProdutos

ProdutoController --> Rota : HTTP 200 OK + JSON(listaProdutos)

Rota --> Cliente : exibirProdutos()

@enduml
